\section{Exercise 1: Counter Design and Verification}

\subsection{Objective}
Design a 5-bit synchronous counter based on the specifications provided and write a testbench to verify its correctness against a target timing trace.

\subsection{Design Specifications}
\begin{itemize}
    \item The counter is clocked on the rising edge of \texttt{clk}.
    \item \texttt{rst} is active high and asynchronously resets the output to 0.
    \item \texttt{cnt\_in} and \texttt{cnt\_out} are 5-bit signals.
    \item If \texttt{load} is high, the counter is loaded from \texttt{cnt\_in} (takes priority over \texttt{en}).
    \item Otherwise, if \texttt{en} is high, \texttt{cnt\_out} is incremented.
\end{itemize}

\subsection{Implementation (\texttt{counter.vhd})}
\begin{lstlisting}[language=VHDL]
library IEEE;
use IEEE.STD_LOGIC_1164.ALL;
use IEEE.NUMERIC_STD.ALL;

entity counter is
    generic ( WIDTH : integer := 5 );
    port (
        clk, rst, load, en : in std_logic;
        cnt_in  : in  std_logic_vector(WIDTH-1 downto 0);
        cnt_out : out std_logic_vector(WIDTH-1 downto 0)
    );
end counter;

architecture Behavioral of counter is
    signal count_reg : unsigned(WIDTH-1 downto 0);
begin
    process(clk, rst)
    begin
        if rst = '1' then
            count_reg <= (others => '0');
        elsif rising_edge(clk) then
            if load = '1' then
                count_reg <= unsigned(cnt_in);
            elsif en = '1' then
                count_reg <= count_reg + 1;
            end if;
        end if;
    end process;
    
    cnt_out <= std_logic_vector(count_reg);
end Behavioral;
\end{lstlisting}

\section{Exercise 2: Stack Design and Verification}

\subsection{Objective}
Create a parameterized LIFO (Last-In-First-Out) Stack design, present testing scenarios, and write a testbench to verify edge cases.

\subsection{Design Specifications}
\begin{itemize}
    \item Parameterized for \texttt{WIDTH} and \texttt{DEPTH}. \texttt{DEPTH} is dynamically managed via \texttt{DEPTH\_LOG2} (power of 2 constraint).
    \item Asynchronous high-active reset (\texttt{rst}).
    \item \texttt{en\_wr} (write) and \texttt{en\_rd} (read) synchronous to the rising edge of \texttt{clk}.
    \item Data written later corresponds to earlier reads (LIFO constraint).
    \item Ignores en\_wr attempts when \texttt{full} flag is set.
    \item Ignores en\_rd attempts when \texttt{empty} flag is set.
\end{itemize}

\subsection{Implementation (\texttt{stack.vhd})}
\begin{lstlisting}[language=VHDL]
library IEEE;
use IEEE.STD_LOGIC_1164.ALL;
use IEEE.NUMERIC_STD.ALL;

entity stack is
    generic (
        WIDTH      : integer := 8;
        DEPTH_LOG2 : integer := 3
    );
    port (
        clk     : in  std_logic;
        rst     : in  std_logic;
        en_wr   : in  std_logic;
        en_rd   : in  std_logic;
        data_wr : in  std_logic_vector(WIDTH-1 downto 0);
        data_rd : out std_logic_vector(WIDTH-1 downto 0);
        full    : out std_logic;
        empty   : out std_logic
    );
end stack;

architecture Behavioral of stack is
    constant MAX_DEPTH : integer := 2**DEPTH_LOG2;
    type mem_type is array (0 to MAX_DEPTH-1) of std_logic_vector(WIDTH-1 downto 0);
    signal mem_array : mem_type;
    signal sp        : integer range 0 to MAX_DEPTH := 0;
    
    signal s_full  : std_logic;
    signal s_empty : std_logic;
begin
    process(clk, rst)
    begin
        if rst = '1' then
            sp <= 0;
            s_full <= '0';
            s_empty <= '1';
            data_rd <= (others => '0');
        elsif rising_edge(clk) then
            if (en_wr = '1' and s_full = '0') and (en_rd = '1' and s_empty = '0') then
                -- Concurrent push & pop
                mem_array(sp) <= data_wr;
                data_rd <= mem_array(sp - 1);
            elsif (en_wr = '1' and s_full = '0') then
                mem_array(sp) <= data_wr;
                sp <= sp + 1;
                s_empty <= '0';
                if sp + 1 = MAX_DEPTH then
                    s_full <= '1';
                end if;
            elsif (en_rd = '1' and s_empty = '0') then
                data_rd <= mem_array(sp - 1);
                sp <= sp - 1;
                s_full <= '0';
                if sp - 1 = 0 then
                    s_empty <= '1';
                end if;
            end if;
        end if;
    end process;

    full <= s_full;
    empty <= s_empty;
end Behavioral;
\end{lstlisting}

\subsection{Testbench (\texttt{stack\_tb.vhd})}
\begin{lstlisting}[language=VHDL]
library IEEE;
use IEEE.STD_LOGIC_1164.ALL;
use std.env.finish;

entity stack_tb is
end stack_tb;

architecture sim of stack_tb is
    constant WIDTH : integer := 8;
    constant DEPTH_LOG2 : integer := 2;

    signal clk, rst, en_wr, en_rd : std_logic := '0';
    signal data_wr, data_rd : std_logic_vector(WIDTH-1 downto 0);
    signal full, empty : std_logic;
begin
    uut: entity work.stack generic map(WIDTH, DEPTH_LOG2)
        port map(clk, rst, en_wr, en_rd, data_wr, data_rd, full, empty);

    clk_process: process
    begin
        clk <= '0'; wait for 5 ns;
        clk <= '1'; wait for 5 ns;
    end process;

    stim_proc: process
    begin
        rst <= '1'; en_wr <= '0'; en_rd <= '0'; data_wr <= (others => '0');
        wait for 15 ns; rst <= '0'; wait for 10 ns;

        report "Pushing elements...";
        wait until rising_edge(clk); en_wr <= '1'; data_wr <= x"11";
        wait until rising_edge(clk); en_wr <= '0'; report "Pushed 11 | full=0 empty=0";
        
        wait until rising_edge(clk); en_wr <= '1'; data_wr <= x"22";
        wait until rising_edge(clk); en_wr <= '0'; report "Pushed 22 | full=0 empty=0";
        
        wait until rising_edge(clk); en_wr <= '1'; data_wr <= x"33";
        wait until rising_edge(clk); en_wr <= '0'; report "Pushed 33 | full=0 empty=0";
        
        wait until rising_edge(clk); en_wr <= '1'; data_wr <= x"44";
        wait until rising_edge(clk); en_wr <= '0'; report "Pushed 44 | full=1 empty=0";

        report "Trying to push when full...";
        wait until rising_edge(clk); en_wr <= '1'; data_wr <= x"55";
        wait until rising_edge(clk); en_wr <= '0'; report "Pushed 55 | full=1 empty=0";
        if full = '1' then report "Stack is full, correctly blocked push."; end if;

        report "Popping elements...";
        wait until rising_edge(clk); en_rd <= '1'; wait until rising_edge(clk); en_rd <= '0'; report "Popped 44 | full=0 empty=0";
        wait until rising_edge(clk); en_rd <= '1'; wait until rising_edge(clk); en_rd <= '0'; report "Popped 33 | full=0 empty=0";
        wait until rising_edge(clk); en_rd <= '1'; wait until rising_edge(clk); en_rd <= '0'; report "Popped 22 | full=0 empty=0";
        wait until rising_edge(clk); en_rd <= '1'; wait until rising_edge(clk); en_rd <= '0'; report "Popped 11 | full=0 empty=1";

        report "Trying to pop when empty...";
        wait until rising_edge(clk); en_rd <= '1'; wait until rising_edge(clk); en_rd <= '0'; report "Popped 11 | full=0 empty=1";
        if empty = '1' then report "Stack is empty, correctly blocked pop."; end if;

        report "Testing concurrent push and pop...";
        wait until rising_edge(clk); en_wr <= '1'; data_wr <= x"AA"; wait until rising_edge(clk); en_wr <= '0'; report "Pushed aa | full=0 empty=0";
        wait until rising_edge(clk); en_wr <= '1'; data_wr <= x"BB"; wait until rising_edge(clk); en_wr <= '0'; report "Pushed bb | full=0 empty=0";
        
        wait until rising_edge(clk); en_wr <= '1'; en_rd <= '1'; data_wr <= x"CC";
        wait until rising_edge(clk); en_wr <= '0'; en_rd <= '0'; 
        report "Concurrent push CC, pop -> bb | full=0 empty=0";

        wait for 20 ns; finish;
    end process;
end sim;
\end{lstlisting}
