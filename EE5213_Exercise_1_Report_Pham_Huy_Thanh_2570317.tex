\section{Exercise 1: Counter Design and Verification}

\subsection{Objective}
Design a 5-bit synchronous counter based on the specifications provided and write a testbench to verify its correctness against a target timing trace.

\subsection{Design Specifications}
\begin{itemize}
    \item The counter is clocked on the rising edge of \texttt{clk}.
    \item \texttt{rst} is active high and asynchronously resets the output to 0.
    \item \texttt{cnt\_in} and \texttt{cnt\_out} are 5-bit signals.
    \item If \texttt{load} is high, the counter is loaded from \texttt{cnt\_in} (takes priority over \texttt{en}).
    \item Otherwise, if \texttt{en} is high, \texttt{cnt\_out} is incremented.
\end{itemize}

\subsection{Implementation (\texttt{counter.v})}
\begin{lstlisting}[language=Verilog]
`timescale 1ns / 1ps

module counter (
    input clk,
    input rst,
    input load,
    input en,
    input [4:0] cnt_in,
    output reg [4:0] cnt_out
);
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            cnt_out <= 5'b00000;
        end else if (load) begin
            cnt_out <= cnt_in;
        end else if (en) begin
            cnt_out <= cnt_out + 1;
        end
    end
endmodule
\end{lstlisting}

\subsection{Testbench (\texttt{counter\_tb.v})}
To match the precise timestamps detailed in the exercise, the testbench drives inputs securely synchronized around the clock edges.

\begin{lstlisting}[language=Verilog]

\section{Exercise 2: Stack Design and Verification}

\subsection{Objective}
Create a parameterized LIFO (Last-In-First-Out) Stack design, present testing scenarios, and write a testbench to verify edge cases.

\subsection{Design Specifications}
\begin{itemize}
    \item Parameterized for \texttt{WIDTH} and \texttt{DEPTH}. \texttt{DEPTH} is dynamically managed via \texttt{DEPTH\_LOG2} (power of 2 constraint).
    \item Asynchronous high-active reset (\texttt{rst}).
    \item \texttt{en\_wr} (write) and \texttt{en\_rd} (read) synchronous to the rising edge of \texttt{clk}.
    \item Data written later corresponds to earlier reads (LIFO constraint).
    \item Ignores en\_wr attempts when \texttt{full} flag is set.
    \item Ignores en\_rd attempts when \texttt{empty} flag is set.
\end{itemize}

\subsection{Implementation (\texttt{stack.v})}
\begin{lstlisting}[language=Verilog]
`timescale 1ns / 1ps

module stack #(
    parameter WIDTH = 8,
    parameter DEPTH_LOG2 = 3
) (
    input clk, rst, en_wr, en_rd,
    input [WIDTH-1:0] data_wr,
    output reg [WIDTH-1:0] data_rd,
    output reg full, empty
);
    localparam MAX_DEPTH = 1 << DEPTH_LOG2;
    reg [WIDTH-1:0] mem [0:MAX_DEPTH-1];
    reg [DEPTH_LOG2:0] sp;

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            sp <= 0; full <= 0; empty <= 1; data_rd <= 0;
        end else begin
            if (en_wr && !full && en_rd && !empty) begin
                // Synchronous concurrent push & pop
                // Keeps stack pointer, full, empty unchanged
                mem[sp] <= data_wr;
                data_rd <= mem[sp - 1];
            end else if (en_wr && !full) begin
                mem[sp] <= data_wr;
                sp <= sp + 1;
                empty <= 0;
                if (sp + 1 == MAX_DEPTH) full <= 1;
            end else if (en_rd && !empty) begin
                data_rd <= mem[sp - 1];
                sp <= sp - 1;
                full <= 0;
                if (sp - 1 == 0) empty <= 1;
            end
        end
    end
endmodule
\end{lstlisting}

\subsection{Testbench (\texttt{stack\_tb.v})}
\begin{lstlisting}[language=Verilog]
`timescale 1ns / 1ps

module stack_tb;
    parameter WIDTH = 8;
    parameter DEPTH_LOG2 = 2; // Testing depth 4

    reg clk, rst, en_wr, en_rd;
    reg [WIDTH-1:0] data_wr;
    wire [WIDTH-1:0] data_rd;
    wire full, empty;

    stack #(.WIDTH(WIDTH), .DEPTH_LOG2(DEPTH_LOG2)) uut (
        .clk(clk), .rst(rst), .en_wr(en_wr), .en_rd(en_rd), 
        .data_wr(data_wr), .data_rd(data_rd), .full(full), .empty(empty)
    );

    initial forever #5 clk = ~clk;

    initial begin
        rst = 1; en_wr = 0; en_rd = 0; data_wr = 0;
        #15; rst = 0; #10;

        $display("Pushing elements...");
        push_data(8'h11); push_data(8'h22); push_data(8'h33); push_data(8'h44);

        $display("Trying to push when full...");
        push_data(8'h55);
        if (full) $display("Stack is full, correctly blocked push.");

        $display("Popping elements...");
        pop_data(); pop_data(); pop_data(); pop_data();

        $display("Trying to pop when empty...");
        pop_data();
        if (empty) $display("Stack is empty, correctly blocked pop.");

        $display("Testing concurrent push and pop...");
        push_data(8'hAA);
        push_data(8'hBB);
        @(posedge clk); en_wr = 1; en_rd = 1; data_wr = 8'hCC;
        @(posedge clk); en_wr = 0; en_rd = 0;
        $display("Concurrent push CC, pop -> %h | full=%b empty=%b", data_rd, full, empty);

        #20 $finish;
    end

    task push_data(input [WIDTH-1:0] val);
    begin
        @(posedge clk); en_wr = 1; data_wr = val;
        @(posedge clk); en_wr = 0;
        $display("Pushed %h | full=%b empty=%b", val, full, empty);
    end
    endtask

    task pop_data();
    begin
        @(posedge clk); en_rd = 1;
        @(posedge clk); en_rd = 0;
        $display("Popped %h | full=%b empty=%b", data_rd, full, empty);
    end
    endtask
endmodule
\end{lstlisting}
