\section{Exercise 1: Counter Design and Verification}

\subsection{Objective}
Design a 5-bit synchronous counter based on the specifications provided and write a testbench to verify its correctness against a target timing trace.

\subsection{Design Specifications}
\begin{itemize}
    \item The counter is clocked on the rising edge of \texttt{clk}.
    \item \texttt{rst} is active high and asynchronously resets the output to 0.
    \item \texttt{cnt\_in} and \texttt{cnt\_out} are 5-bit signals.
    \item If \texttt{load} is high, the counter is loaded from \texttt{cnt\_in} (takes priority over \texttt{enab}).
    \item Otherwise, if \texttt{enab} is high, \texttt{cnt\_out} is incremented.
\end{itemize}

\subsection{Implementation (\texttt{counter.v})}
\begin{lstlisting}[language=Verilog]
`timescale 1ns / 1ps

module counter (
    input clk,
    input rst,
    input load,
    input enab,
    input [4:0] cnt_in,
    output reg [4:0] cnt_out
);
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            cnt_out <= 5'b00000;
        end else if (load) begin
            cnt_out <= cnt_in;
        end else if (enab) begin
            cnt_out <= cnt_out + 1;
        end
    end
endmodule
\end{lstlisting}

\subsection{Testbench (\texttt{counter\_tb.v})}

\section{Exercise 2: Stack Design and Verification}

\subsection{Objective}
Create a parameterized LIFO (Last-In-First-Out) Stack design, present testing scenarios, and write a testbench to verify edge cases.

\subsection{Design Specifications}
\begin{itemize}
    \item Parameterized for \texttt{WIDTH} and \texttt{DEPTH}. \texttt{DEPTH} is dynamically managed via \texttt{DEPTH\_LOG2} (power of 2 constraint).
    \item Asynchronous high-active reset (\texttt{rst}).
    \item \texttt{push} (write) and \texttt{pop} (read) synchronous to the rising edge of \texttt{clk}.
    \item Data written later corresponds to earlier reads (LIFO constraint).
    \item Ignores push attempts when \texttt{full} flag is set.
    \item Ignores pop attempts when \texttt{empty} flag is set.
\end{itemize}

\subsection{Implementation (\texttt{stack.v})}
\begin{lstlisting}[language=Verilog]
`timescale 1ns / 1ps

module stack #(
    parameter WIDTH = 8,
    parameter DEPTH_LOG2 = 3
) (
    input clk, rst, push, pop,
    input [WIDTH-1:0] din,
    output reg [WIDTH-1:0] dout,
    output reg full, empty
);
    localparam MAX_DEPTH = 1 << DEPTH_LOG2;
    reg [WIDTH-1:0] mem [0:MAX_DEPTH-1];
    reg [DEPTH_LOG2:0] sp;

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            sp <= 0; full <= 0; empty <= 1; dout <= 0;
        end else begin
            if (push && !full) begin
                mem[sp] <= din;
                sp <= sp + 1;
                empty <= 0;
                if (sp + 1 == MAX_DEPTH) full <= 1;
            end else if (pop && !empty) begin
                dout <= mem[sp - 1];
                sp <= sp - 1;
                full <= 0;
                if (sp - 1 == 0) empty <= 1;
            end
        end
    end
endmodule
\end{lstlisting}

\subsection{Testbench (\texttt{stack\_tb.v})}
\begin{lstlisting}[language=Verilog]

    parameter DEPTH_LOG2 = 2; // Testing depth 4

    reg clk, rst, push, pop;
    reg [WIDTH-1:0] din;
    wire [WIDTH-1:0] dout;
    wire full, empty;

    stack #(.WIDTH(WIDTH), .DEPTH_LOG2(DEPTH_LOG2)) uut (
        .clk(clk), .rst(rst), .push(push), .pop(pop), 
        .din(din), .dout(dout), .full(full), .empty(empty)
    );

    initial forever #5 clk = ~clk;

    initial begin
        rst = 1; push = 0; pop = 0; din = 0;
        #15; rst = 0; #10;

        $display("Pushing elements...");
        push_data(8'h11); push_data(8'h22); push_data(8'h33); push_data(8'h44);

        $display("Trying to push when full...");
        push_data(8'h55);
        if (full) $display("Stack is full, correctly blocked push.");

        $display("Popping elements...");
        pop_data(); pop_data(); pop_data(); pop_data();

        $display("Trying to pop when empty...");
        pop_data();
        if (empty) $display("Stack is empty, correctly blocked pop.");

        #20 $finish;
    end

    task push_data(input [WIDTH-1:0] val);
    begin
        @(posedge clk); push = 1; din = val;
        @(posedge clk); push = 0;
        $display("Pushed %h | full=%b empty=%b", val, full, empty);
    end
    endtask

    task pop_data();
    begin
        @(posedge clk); pop = 1;
        @(posedge clk); pop = 0;
        $display("Popped %h | full=%b empty=%b", dout, full, empty);
    end
    endtask
endmodule

\section{Exercise 3: FSM Reachability Analysis}

\subsection{Objective}
Implement the State Reachability Analysis algorithm for an arbitrary Finite State Machine (FSM) using C++. The program computes the set of reachable states given a set of initial states and a transition relation.

\subsection{Design Specifications}
\begin{itemize}
    \item \textbf{Input:} Set of initial states ($I$), and a transition relation ($R$) mapping current states to next possible states.
    \item \textbf{Output:} The complete set of reachable states ($ARS$ - All Reached States).
    \item \textbf{Algorithm Strategy:}
    \begin{itemize}
        \item Track \textbf{All Reached States} (\texttt{ARS}) and \textbf{New Next States} (\texttt{NNS} or Frontier Set).
        \item Iteratively compute the next states (\texttt{NS}) from the current frontier \texttt{NNS}.
        \item Update the frontier as newly discovered states (\texttt{NNS} = \texttt{NS} - \texttt{ARS}).
        \item Add the new frontier to the all reached states (\texttt{ARS} = \texttt{ARS} $\cup$ \texttt{NNS}).
        \item Terminate when the frontier is empty.
    \end{itemize}
\end{itemize}

\subsection{Implementation (\texttt{fsm\_reachability.cpp})}
\begin{lstlisting}[language=C++]
#include <iostream>
#include <vector>
#include <set>
#include <map>
#include <algorithm>

typedef int State;

std::set<State> Reachability(
    const std::map<State, std::set<State>>& R,
    const std::set<State>& I
) {
    std::set<State> ARS = I;  // All Reached States
    std::set<State> NNS = I;  // New Next States (Frontier Set)

    while (true) {
        std::set<State> NS; 
        for (State s : NNS) {
            if (R.find(s) != R.end()) {
                for (State next_s : R.at(s)) {
                    NS.insert(next_s);
                }
            }
        }

        std::set<State> new_NNS;
        std::set_difference(
            NS.begin(), NS.end(),
            ARS.begin(), ARS.end(),
            std::inserter(new_NNS, new_NNS.begin())
        );
        NNS = new_NNS;

        if (NNS.empty()) {
            break;
        }
        ARS.insert(NNS.begin(), NNS.end());
    }
    return ARS;
}

int main() {
    std::map<State, std::set<State>> transition_relation = {
        {0, {1, 5}}, {1, {2}}, {5, {5, 2}},
        {2, {3}}, {3, {4}}, {4, {}}
    };
    std::set<State> initial_states = {0};
    std::set<State> reachable_states = Reachability(transition_relation, initial_states);

    std::cout << "Final Reachable States (ARS): { ";
    for (State s : reachable_states) std::cout << "S" << s << " ";
    std::cout << "}\n";
    return 0;
}
\end{lstlisting}
