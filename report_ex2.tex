\section{Exercise 2: FSM Reachability Analysis}

\subsection{Objective}
Implement the State Reachability Analysis algorithm for an arbitrary Finite State Machine (FSM) using C++. The program computes the set of reachable states given a set of initial states and a transition relation.

\subsection{Design Specifications}
\begin{itemize}
    \item \textbf{Input:} Set of initial states ($I$), and a transition relation ($R$) mapping current states to next possible states.
    \item \textbf{Output:} The complete set of reachable states ($ARS$ - All Reached States).
    \item \textbf{Algorithm Strategy:}
    \begin{itemize}
        \item Track \textbf{All Reached States} (\texttt{ARS}) and \textbf{New Next States} (\texttt{NNS} or Frontier Set).
        \item Iteratively compute the next states (\texttt{NS}) from the current frontier \texttt{NNS}.
        \item Update the frontier as newly discovered states (\texttt{NNS} = \texttt{NS} - \texttt{ARS}).
        \item Add the new frontier to the all reached states (\texttt{ARS} = \texttt{ARS} $\cup$ \texttt{NNS}).
        \item Terminate when the frontier is empty.
    \end{itemize}
\end{itemize}

\subsection{Example FSM Analysis}

To illustrate the algorithm, consider the FSM defined in the implementation with the following transition relations:
\begin{itemize}
    \item $S_0 \rightarrow \{S_1, S_5\}$
    \item $S_1 \rightarrow \{S_2\}$
    \item $S_5 \rightarrow \{S_5, S_2\}$
    \item $S_2 \rightarrow \{S_3\}$
    \item $S_3 \rightarrow \{S_4\}$
    \item $S_4 \rightarrow \emptyset$
\end{itemize}

The initial state set is $I = \{S_0\}$. The state diagram is shown in Figure \ref{fig:fsm}.

\textbf{Step-by-Step Execution:}
\begin{enumerate}
    \item \textbf{Initialization:}
    \begin{itemize}
        \item $ARS = \{S_0\}$
        \item Frontier $NNS = \{S_0\}$
    \end{itemize}
    \item \textbf{Iteration 1:}
    \begin{itemize}
        \item Compute next states from $NNS$: $NS = \{S_1, S_5\}$
        \item Compute new frontier: $NNS = NS \setminus ARS = \{S_1, S_5\} \setminus \{S_0\} = \{S_1, S_5\}$
        \item Update $ARS$: $ARS = ARS \cup NNS = \{S_0, S_1, S_5\}$
    \end{itemize}
    \item \textbf{Iteration 2:}
    \begin{itemize}
        \item Compute next states from $NNS$: $NS = \{S_2, S_5\}$ (from $S_1 \rightarrow S_2$ and $S_5 \rightarrow S_2, S_5$)
        \item Compute new frontier: $NNS = NS \setminus ARS = \{S_2, S_5\} \setminus \{S_0, S_1, S_5\} = \{S_2\}$
        \item Update $ARS$: $ARS = ARS \cup NNS = \{S_0, S_1, S_2, S_5\}$
    \end{itemize}
    \item \textbf{Iteration 3:}
    \begin{itemize}
        \item Compute next states from $NNS$: $NS = \{S_3\}$ (from $S_2 \rightarrow S_3$)
        \item Compute new frontier: $NNS = NS \setminus ARS = \{S_3\} \setminus \{S_0, S_1, S_2, S_5\} = \{S_3\}$
        \item Update $ARS$: $ARS = ARS \cup NNS = \{S_0, S_1, S_2, S_3, S_5\}$
    \end{itemize}
    \item \textbf{Iteration 4:}
    \begin{itemize}
        \item Compute next states from $NNS$: $NS = \{S_4\}$ (from $S_3 \rightarrow S_4$)
        \item Compute new frontier: $NNS = NS \setminus ARS = \{S_4\} \setminus \{S_0, S_1, S_2, S_3, S_5\} = \{S_4\}$
        \item Update $ARS$: $ARS = ARS \cup NNS = \{S_0, S_1, S_2, S_3, S_4, S_5\}$
    \end{itemize}
    \item \textbf{Iteration 5:}
    \begin{itemize}
        \item Compute next states from $NNS$: $NS = \emptyset$ (from $S_4 \rightarrow \emptyset$)
        \item Compute new frontier: $NNS = NS \setminus ARS = \emptyset \setminus ARS = \emptyset$
        \item The frontier $NNS$ is empty, so the algorithm terminates.
    \end{itemize}
\end{enumerate}

\textbf{Final Result:} $ARS = \{S_0, S_1, S_2, S_3, S_4, S_5\}$

\subsection{Implementation (\texttt{fsm\_reachability.cpp})}
\begin{lstlisting}[language=C++]
#include <iostream>
#include <vector>
#include <set>
#include <map>
#include <algorithm>

typedef int State;

std::set<State> Reachability(
    const std::map<State, std::set<State>>& R,
    const std::set<State>& I
) {
    std::set<State> ARS = I;  // All Reached States
    std::set<State> NNS = I;  // New Next States (Frontier Set)

    while (true) {
        std::set<State> NS; 
        for (State s : NNS) {
            if (R.find(s) != R.end()) {
                for (State next_s : R.at(s)) {
                    NS.insert(next_s);
                }
            }
        }

        std::set<State> new_NNS;
        std::set_difference(
            NS.begin(), NS.end(),
            ARS.begin(), ARS.end(),
            std::inserter(new_NNS, new_NNS.begin())
        );
        NNS = new_NNS;

        if (NNS.empty()) {
            break;
        }
        ARS.insert(NNS.begin(), NNS.end());
    }
    return ARS;
}

int main() {
    std::map<State, std::set<State>> transition_relation = {
        {0, {1, 5}}, {1, {2}}, {5, {5, 2}},
        {2, {3}}, {3, {4}}, {4, {}}
    };
    std::set<State> initial_states = {0};
    std::set<State> reachable_states = Reachability(transition_relation, initial_states);

    std::cout << "Final Reachable States (ARS): { ";
    for (State s : reachable_states) std::cout << "S" << s << " ";
    std::cout << "}\n";
    return 0;
}
\end{lstlisting}
